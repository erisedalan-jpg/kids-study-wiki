"""考点名语义近义归并检测：词法相似度扫描 canonical 规范名 → 报告 + 合并草案。
纯函数 + 薄 CLI。绝不改 canonical/词条；合并是人工复核草案后的独立动作。
"""
from __future__ import annotations

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[0]))

from _utils import setup_utf8  # noqa: E402

# ── 去噪：删除不影响语义的虚词，使「函数的单调性」≡「函数单调性」 ──
_NOISE = ("的", "之")
_TAIL = ("运算", "问题")


def _denoise(s: str) -> str:
    for ch in _NOISE:
        s = s.replace(ch, "")
    return s.strip()


def _strip_tail(s: str) -> str:
    for t in _TAIL:
        if s.endswith(t) and len(s) > len(t):
            s = s[: -len(t)]
    return s


def _bigrams(s: str) -> set[str]:
    if len(s) < 2:
        return {s} if s else set()
    return {s[i:i + 2] for i in range(len(s) - 1)}


def s_bigram(a: str, b: str) -> float:
    ba, bb = _bigrams(a), _bigrams(b)
    if not ba or not bb:
        return 0.0
    return len(ba & bb) / len(ba | bb)


def s_jaccard(a: str, b: str) -> float:
    ca, cb = set(a), set(b)
    if not ca or not cb:
        return 0.0
    return len(ca & cb) / len(ca | cb)


def _lev(a: str, b: str) -> int:
    if a == b:
        return 0
    if not a:
        return len(b)
    if not b:
        return len(a)
    prev = list(range(len(b) + 1))
    for i, ca in enumerate(a, 1):
        cur = [i]
        for j, cb in enumerate(b, 1):
            cost = 0 if ca == cb else 1
            cur.append(min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + cost))
        prev = cur
    return prev[-1]


def s_edit(a: str, b: str) -> float:
    m = max(len(a), len(b))
    return 1.0 - _lev(a, b) / m if m else 0.0


def s_contain(a: str, b: str) -> float:
    """a,b 已去噪。去尾缀后相等或互为子串 → 0.95。"""
    na, nb = _strip_tail(a), _strip_tail(b)
    if not na or not nb:
        return 0.0
    if na == nb or na in nb or nb in na:
        return 0.95
    return 0.0


def name_similarity(a: str, b: str) -> float:
    """词法相似度 ∈[0,1]。可插拔接口——嵌入层日后替换此函数。"""
    if a == b:
        return 1.0
    na, nb = _denoise(a), _denoise(b)
    if na == nb:
        return 0.97
    weighted = 0.6 * s_bigram(na, nb) + 0.25 * s_jaccard(na, nb) + 0.15 * s_edit(na, nb)
    return max(s_contain(na, nb), weighted)
```

import yaml  # noqa: E402


def load_contrast_groups(subject: str) -> list[set[str]]:
    """读 contrast_terms.yaml，返回 通用 + 该科 的词组（set 列表）。"""
    path = Path(__file__).parent / "contrast_terms.yaml"
    data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
    groups: list[set[str]] = []
    for key in ("通用", subject):
        for grp in data.get(key, []) or []:
            if grp:
                groups.append(set(grp))
    return groups


def is_contrast_pair(a: str, b: str, groups: list[set[str]]) -> bool:
    """a 把某组内一词换成同组另一词后 == b → 对立项。差一个对立词即命中。"""
    for grp in groups:
        for t1 in grp:
            if t1 in a:
                for t2 in grp:
                    if t2 != t1 and a.replace(t1, t2) == b:
                        return True
    return False
```

```python
def load_canonical_map(subject: str) -> dict[str, str]:
    """读 canonical_考点_<科>.yaml（变体→规范名）。无表则报错退出。"""
    path = Path(__file__).parent / f"canonical_考点_{subject}.yaml"
    if not path.exists():
        raise SystemExit(f"无 canonical 表（非 4 理科？）：{subject}")
    return yaml.safe_load(path.read_text(encoding="utf-8")) or {}


def load_canonical_names(subject: str) -> set[str]:
    """canonical 表的规范名（RHS）去重集——去重宇宙。"""
    return set(load_canonical_map(subject).values())


def load_parent_map(subject: str) -> dict[str, str]:
    """规范名 → 父主题；未列入由调用方归「其他」。"""
    import gen_exam_blueprint as g
    _, _, group_yaml, _ = g.paths_for(subject)
    return g.load_group(group_yaml) if group_yaml.exists() else {}


def stats_from_rows(rows: list[dict], normalize: dict, group: dict) -> dict[str, dict]:
    """纯函数：真题行 → {规范名: {真题数,题位,父主题,...}}。行内考点先 parse + canon 归一，
    并按行去重（同行多个变体若归同一规范名，只计一次，防 build_kaodian_map 重复 append）。"""
    import gen_exam_blueprint as g
    from kaodian_aggregate import build_kaodian_map
    atoms = []
    for r in rows:
        kps = sorted({g.canon(k, normalize) for k in g.parse_kaodian(r.get("考点", ""))})
        a = dict(r)
        a["考点"] = kps
        atoms.append(a)
    return build_kaodian_map(atoms, group)


def load_exam_stats(subject: str) -> dict[str, dict]:
    """读吉林卷真题 → 规范名统计（缺真题的规范名不在内，调用方默认 0）。
    用 canonical 表归一（真题已迁移规范名，canon 等幂/兜底零散变体）。"""
    import gen_exam_blueprint as g
    exam_dir, _, group_yaml, _ = g.paths_for(subject)
    rows = g.read_rows(exam_dir) if exam_dir.exists() else []
    normalize = load_canonical_map(subject)
    group = g.load_group(group_yaml) if group_yaml.exists() else {}
    return stats_from_rows(rows, normalize, group)
```

    parent: str
    is_contrast: bool


def find_candidates(names, parent_map, score_fn, th_review, groups, cross_parent=False):
    """同父主题桶内 all-pairs 打分；score≥th_review 或对立项即收集。"""
    names = sorted(set(n for n in names if n))
    out: list[Candidate] = []

    def consider(x: str, y: str) -> None:
        sc = score_fn(x, y)
        contrast = is_contrast_pair(x, y, groups)
        if contrast or sc >= th_review:
            out.append(Candidate(x, y, sc, parent_map.get(x, "其他"), contrast))

    if cross_parent:
        for i in range(len(names)):
            for j in range(i + 1, len(names)):
                consider(names[i], names[j])
    else:
        buckets: dict[str, list[str]] = defaultdict(list)
        for n in names:
            buckets[parent_map.get(n, "其他")].append(n)
        for bucket in buckets.values():
            for i in range(len(bucket)):
                for j in range(i + 1, len(bucket)):
                    consider(bucket[i], bucket[j])
    return out


def cluster(candidates, th_high):
    """高分非对立边并查集聚簇，返回 ≥2 成员的簇列表。"""
    parent: dict[str, str] = {}

    def find(x: str) -> str:
        parent.setdefault(x, x)
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    for c in candidates:
        if c.score >= th_high and not c.is_contrast:
            parent.setdefault(c.a, c.a)
            parent.setdefault(c.b, c.b)
            parent[find(c.a)] = find(c.b)
    groups: dict[str, set] = defaultdict(set)
    for x in list(parent):
        groups[find(x)].add(x)
    return [s for s in groups.values() if len(s) >= 2]


def choose_target(cluster_set, stats):
    """簇内选规范名：真题数最多 → 名最短 → 字典序。"""
    def key(n: str):
        return (-int(stats.get(n, {}).get("真题数", 0)), len(n), n)
    return sorted(cluster_set, key=key)[0]


def build_proposed_merges(clusters, stats):
    """簇内非目标成员 → 目标，组成 变体: 规范名 草案。"""
    merges: dict[str, str] = {}
    for cl in clusters:
        target = choose_target(cl, stats)
        for member in sorted(cl):
            if member != target:
                merges[member] = target
    return merges
```

def _exam_n(stats: dict, name: str) -> int:
    return int(stats.get(name, {}).get("真题数", 0))


def render_report(subject, clusters, review_pairs, contrast_pairs, stats, date_str):
    """三区报告 markdown：高置信合并簇 / 待复核对 / 对立项警示。"""
    L = [f"---\ntype: meta\ntitle: \"考点去重报告 {subject} {date_str}\"\n---\n",
         f"# 考点去重报告 · {subject} · {date_str}\n",
         (f"摘要：高置信簇 {len(clusters)} · 待复核对 {len(review_pairs)} · "
          f"对立警示 {len(contrast_pairs)}\n")]

    L.append("## 高置信合并簇（建议并入 canonical）\n")
    if not clusters:
        L.append("（无）\n")
    for cl in clusters:
        target = choose_target(cl, stats)
        L.append(f"- **目标：{target}**（真题 {_exam_n(stats, target)}）")
        for m in sorted(cl):
            if m != target:
                L.append(f"  - {m}（真题 {_exam_n(stats, m)}）")

    L.append("\n## 待复核对（人工判断是否同义）\n")
    if not review_pairs:
        L.append("（无）\n")
    for c in sorted(review_pairs, key=lambda x: -x.score):
        L.append(f"- {c.a} / {c.b} — score {c.score:.2f} · 父主题 {c.parent}")

    L.append("\n## ⚠️ 对立项警示（差对立词，勿并）\n")
    if not contrast_pairs:
        L.append("（无）\n")
    for c in contrast_pairs:
        L.append(f"- {c.a} / {c.b} — 命中对立组 · 父主题 {c.parent}")

    return "\n".join(L) + "\n"


def render_merges_yaml(merges, subject):
    """合并草案 yaml 文本（含头部注释）。仅高置信非对立项。"""
    head = (f"# {subject} 合并草案——人工删减后将保留行 append 到 "
            f"canonical_考点_{subject}.yaml，\n"
            f"# 再跑标准化 recipe（normalize → exam_index → compute_weight → "
            f"gen_exam_blueprint）。\n"
            f"# 仅含高置信非对立项；绝不含对立项。\n")
    body = "\n".join(f"{k}: {v}" for k, v in sorted(merges.items()))
    return head + body + ("\n" if body else "")
```

def run_subject(subject, th_high, th_review, cross_parent, write):
    """编排单科：加载→候选→聚簇→草案→报告。write=False 仅返回不落盘。"""
    names = load_canonical_names(subject)

    report = render_report(subject, clusters, review, contrast, stats, date_str)
